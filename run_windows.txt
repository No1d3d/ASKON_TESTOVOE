# Сборка с использованием Conan Provider (CMakePresets)
# Conan вызовы скрыты "под капотом" через CMake presets
conan profile detect --force
conan install . --build=missing

# CMake автоматически использует preset conan-release из CMakePresets.json
cmake --preset conan-release
cmake --build . --preset conan-release
ctest --preset conan-release --output-on-failure
